cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
project(NickelApplicationAdapterProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_DISABLE_PRECOMPILE_HEADERS ON)
get_filename_component(NICKEL_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
file(READ "${NICKEL_SOURCE_ROOT}/native/metadata.json" _nickel_metadata)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${NICKEL_SOURCE_ROOT}/native/metadata.json")
string(JSON PROJECT_VERSION GET "${_nickel_metadata}" version)
foreach(_field displayName applicationId executableName license)
    string(JSON _nickel_${_field} GET "${_nickel_metadata}" "${_field}")
endforeach()
configure_file("${NICKEL_SOURCE_ROOT}/native/app/BuildInfo.hpp.in" "${CMAKE_CURRENT_BINARY_DIR}/BuildInfo.hpp" @ONLY)
include("${NICKEL_SOURCE_ROOT}/cmake/NickelDependencies.cmake")
include("${NICKEL_SOURCE_ROOT}/cmake/NickelBuildEnvironment.cmake")
include("${NICKEL_SOURCE_ROOT}/cmake/NickelDomains.cmake")
if(NOT NICKEL_SHARED_NATIVE_ROOT)
    set(NICKEL_SHARED_NATIVE_ROOT "${NICKEL_SOURCE_ROOT}/native")
endif()
add_subdirectory("${NICKEL_SOURCE_ROOT}/native/core/common" common)
target_include_directories(nickel_common BEFORE INTERFACE "${NICKEL_SHARED_NATIVE_ROOT}")
if(MSVC)
    add_compile_options(/utf-8 /EHsc /permissive-)
    add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
endif()
add_subdirectory("${NICKEL_SHARED_NATIVE_ROOT}/core/config" foundation)
add_subdirectory("${NICKEL_SHARED_NATIVE_ROOT}/core/network" transport)
add_subdirectory("${NICKEL_SHARED_NATIVE_ROOT}/core/accounts" identity)
# This is a narrow adapter probe, not a replacement for the full community/root build.
add_library(nickel_app_probe_community STATIC
    "${NICKEL_SHARED_NATIVE_ROOT}/core/resources/types.cpp"
    "${NICKEL_SHARED_NATIVE_ROOT}/core/resources/network.cpp")
target_link_libraries(nickel_app_probe_community PUBLIC nickel_common Threads::Threads)
add_executable(nickel_app_adapter_probe
    probe.cpp
    "${NICKEL_SOURCE_ROOT}/native/app/WorkerQueue.cpp"
    "${NICKEL_SOURCE_ROOT}/native/app/core_adapters.cpp"
    "${NICKEL_SOURCE_ROOT}/native/app/settings.cpp")
target_include_directories(nickel_app_adapter_probe PRIVATE "${NICKEL_SOURCE_ROOT}/native/app" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(nickel_app_adapter_probe PRIVATE nickel_foundation nickel_identity nickel_transport nickel_app_probe_community)
if(MSVC)
    target_compile_options(nickel_app_adapter_probe PRIVATE /W4 /WX)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME app.production-adapters
    COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/run.py" "$<TARGET_FILE:nickel_app_adapter_probe>")
set_tests_properties(app.production-adapters PROPERTIES TIMEOUT 60)
