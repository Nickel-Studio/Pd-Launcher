cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
project(NickelQtRuntimeProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_DISABLE_PRECOMPILE_HEADERS ON)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/NickelBuildEnvironment.cmake")
find_package(Qt6 6.8.3 EXACT CONFIG REQUIRED COMPONENTS Core Gui Widgets Svg Test)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/NickelQtDependencies.cmake")
nickel_configure_qt_dependencies()
if(NOT TARGET Qt6::QWebpPlugin)
    message(FATAL_ERROR "The pinned static Qt WebP plugin is required")
endif()
set(NICKEL_QT_IMAGEFORMATS_SOURCE "" CACHE PATH "Verified QtImageFormats 6.8.3 source with original WebP test fixtures")
set(_fixture_root "${NICKEL_QT_IMAGEFORMATS_SOURCE}/tests/auto/webp/images")
set(_fixture_files kollada.png kollada.webp kollada_lossless.webp kollada_noalpha.webp kollada_animation.webp)
set(_fixture_hashes
    6e34f38886aeee7e50d23413b900fb2c66e9fb8f575e8a13bc650a45645d915b
    570d16de35234f2c184a8369dd6898d1fca086c08fba245eecab2a405589bc72
    038bf8ad5e6c9c140ae7ad2e52735b962ff9acea61043e1ae4adadec9d8f4ae6
    10f260948d63de4b14f8d9b975999ead1c9b92221792fcf2f80ecda3f3bdcf07
    cd063f92789b7fa2ff80fc1e3dde80da13f531747b21ca368f8e7aa2486eed62)
foreach(_index RANGE 0 4)
    list(GET _fixture_files ${_index} _file)
    list(GET _fixture_hashes ${_index} _expected)
    file(SHA256 "${_fixture_root}/${_file}" _actual)
    if(NOT _actual STREQUAL _expected)
        message(FATAL_ERROR "Original locked WebP fixture changed: ${_file}")
    endif()
endforeach()
set(_fixture_root_xml "${_fixture_root}")
string(REPLACE "&" "&amp;" _fixture_root_xml "${_fixture_root_xml}")
string(REPLACE "<" "&lt;" _fixture_root_xml "${_fixture_root_xml}")
string(REPLACE ">" "&gt;" _fixture_root_xml "${_fixture_root_xml}")
configure_file(webp-fixtures.qrc.in "${CMAKE_CURRENT_BINARY_DIR}/webp-fixtures.qrc" @ONLY)
qt_add_executable(nickel_qt_runtime_probe MANUAL_FINALIZATION probe.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/webp-fixtures.qrc")
set_target_properties(nickel_qt_runtime_probe PROPERTIES AUTORCC ON)
target_link_libraries(nickel_qt_runtime_probe PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Svg Qt6::Test)
target_include_directories(nickel_qt_runtime_probe PRIVATE
    "${NICKEL_QT_IMAGEFORMATS_SOURCE}/src/3rdparty/libwebp/src")
qt_import_plugins(nickel_qt_runtime_probe INCLUDE_BY_TYPE imageformats Qt6::QWebpPlugin)
if(WIN32)
    foreach(plugin QWindowsIntegrationPlugin QOffscreenIntegrationPlugin)
        if(NOT TARGET "Qt6::${plugin}")
            message(FATAL_ERROR "Installed static Qt plugin is missing: Qt6::${plugin}")
        endif()
    endforeach()
    qt_import_plugins(nickel_qt_runtime_probe NO_DEFAULT
        INCLUDE Qt6::QWindowsIntegrationPlugin Qt6::QOffscreenIntegrationPlugin)
    target_compile_definitions(nickel_qt_runtime_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
    target_compile_options(nickel_qt_runtime_probe PRIVATE /W4 /WX /utf-8 /EHsc /permissive-)
endif()
qt_finalize_executable(nickel_qt_runtime_probe)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME qt.installed-static-runtime
    COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/run.py" "$<TARGET_FILE:nickel_qt_runtime_probe>")
set_tests_properties(qt.installed-static-runtime PROPERTIES TIMEOUT 60)
