cmake_minimum_required(VERSION 3.24)
set(_community_standalone OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NickelCommunityTests LANGUAGES C CXX)
    set(_community_standalone ON)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
get_filename_component(_community_native_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(_community_shared_native_root "${_community_native_root}")
if(_community_standalone)
    get_filename_component(_worktree "${_community_native_root}/.." ABSOLUTE)
    set(NICKEL_SHARED_NATIVE_ROOT "${_worktree}/.snow/native-shared/native" CACHE PATH "Owner-approved shared source mirror")
    set(_community_shared_native_root "${NICKEL_SHARED_NATIVE_ROOT}")
    if(NOT EXISTS "${_community_shared_native_root}/core/network/http.hpp")
        message(FATAL_ERROR "Supply the actual owner-approved native shared source root")
    endif()
endif()
if(_community_standalone AND MSVC)
    if(POLICY CMP0176)
        cmake_policy(SET CMP0176 NEW)
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(_community_include_probe "${_community_shared_native_root}/tests/foundation/msvc_showincludes.py")
    if(NOT EXISTS "${_community_include_probe}")
        message(FATAL_ERROR "Owner-approved raw MSVC include-prefix probe is required")
    endif()
    execute_process(
        COMMAND "${Python3_EXECUTABLE}" "${_community_include_probe}"
                "${CMAKE_CXX_COMPILER}" "${CMAKE_CURRENT_BINARY_DIR}"
        RESULT_VARIABLE _community_probe_result
        OUTPUT_VARIABLE _community_include_prefix
        ERROR_VARIABLE _community_probe_error
        ENCODING UTF-8)
    if(NOT _community_probe_result EQUAL 0)
        message(FATAL_ERROR "MSVC include-prefix probe failed: ${_community_probe_error}")
    endif()
    string(REGEX REPLACE "[\r\n]+$" "" _community_include_prefix "${_community_include_prefix}")
    set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${_community_include_prefix}")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${_community_include_prefix}")
endif()
if(NOT TARGET nickel_common)
    find_package(nlohmann_json CONFIG REQUIRED)
    add_library(nickel_common INTERFACE)
    target_compile_features(nickel_common INTERFACE cxx_std_20)
    target_include_directories(nickel_common INTERFACE "${_community_native_root}" "${_community_shared_native_root}")
    target_link_libraries(nickel_common INTERFACE nlohmann_json::nlohmann_json)
endif()

option(NICKEL_TEST_CORE_ONLY "Explicit subset without shared HTTP/archive/install/export adapters" OFF)
option(NICKEL_TEST_QT "Test the actual Qt document adapter when the approved static Qt is ready" OFF)
option(NICKEL_TEST_TASK_FAULTS "Instrument the real shared transaction implementation in standalone tests only" OFF)
if(NICKEL_TEST_TASK_FAULTS AND (NOT _community_standalone OR NICKEL_TEST_CORE_ONLY))
    message(FATAL_ERROR "Transaction fault injection is restricted to full standalone tests")
endif()
if(NICKEL_TEST_CORE_ONLY)
    if(NOT _community_standalone)
        message(FATAL_ERROR "The explicit core subset is standalone only; root tests must reuse production targets")
    endif()
    # SecureFile's pinned ancestors use the actual foundation implementation.
    if(NOT TARGET nickel_foundation)
        add_subdirectory("${_community_shared_native_root}/core/config" "${CMAKE_CURRENT_BINARY_DIR}/foundation")
    endif()
    find_package(gumbo CONFIG REQUIRED)
    find_package(tomlplusplus CONFIG REQUIRED)
    add_library(nickel_community_contract_subset STATIC
        "${_community_native_root}/core/resources/types.cpp"
        "${_community_native_root}/core/resources/network.cpp"
        "${_community_native_root}/core/resources/community.cpp"
        "${_community_native_root}/core/resources/mods.cpp"
        "${_community_native_root}/core/resources/utility.cpp"
        "${_community_native_root}/core/resources/file_identity.cpp"
        "${_community_native_root}/core/documents/body.cpp"
        "${_community_shared_native_root}/core/network/hash.cpp"
        "${_community_shared_native_root}/core/network/control.cpp"
        "${_community_shared_native_root}/core/tasks/secure_fs.cpp"
    )
    target_link_libraries(nickel_community_contract_subset PUBLIC nickel_common
        PRIVATE nickel_foundation gumbo::gumbo tomlplusplus::tomlplusplus)
    if(WIN32)
        target_link_libraries(nickel_community_contract_subset PRIVATE bcrypt)
    else()
        find_package(OpenSSL REQUIRED)
        target_link_libraries(nickel_community_contract_subset PRIVATE OpenSSL::Crypto)
    endif()
    add_executable(community_core_tests community_tests.cpp contracts_tests.cpp)
    target_compile_definitions(community_core_tests PRIVATE NICKEL_CORE_ONLY=1)
    target_link_libraries(community_core_tests PRIVATE nickel_community_contract_subset)
    if(MSVC)
        target_compile_options(nickel_community_contract_subset PRIVATE /W4 /permissive- /utf-8 /EHsc)
        set_property(SOURCE "${_community_native_root}/core/documents/body.cpp" APPEND PROPERTY COMPILE_OPTIONS /EHsc-)
        target_compile_options(community_core_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
    endif()
    enable_testing()
    add_test(NAME community_core_contract_subset COMMAND community_core_tests)
    set_tests_properties(community_core_contract_subset PROPERTIES RUN_SERIAL TRUE TIMEOUT 60
        ENVIRONMENT "${_community_test_environment}")
    if(NICKEL_TEST_QT)
        find_package(Qt6 6.8.3 EXACT CONFIG REQUIRED COMPONENTS Core Gui)
        add_library(nickel_community_qt_contract_subset STATIC "${_community_native_root}/core/documents/qt_body.cpp")
        target_link_libraries(nickel_community_qt_contract_subset PUBLIC nickel_community_contract_subset Qt6::Core Qt6::Gui)
        add_executable(community_core_qt_tests qt_tests.cpp)
        target_link_libraries(community_core_qt_tests PRIVATE nickel_community_qt_contract_subset)
        if(MSVC)
            target_compile_options(nickel_community_qt_contract_subset PRIVATE /W4 /permissive- /utf-8 /EHsc)
            target_compile_options(community_core_qt_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
        endif()
        qt_import_plugins(community_core_qt_tests NO_DEFAULT INCLUDE Qt6::QOffscreenIntegrationPlugin Qt6::QJpegPlugin)
        add_test(NAME community_qt_document_subset COMMAND community_core_qt_tests)
        set_tests_properties(community_qt_document_subset PROPERTIES RUN_SERIAL TRUE TIMEOUT 60
            ENVIRONMENT "${_community_test_environment};QT_QPA_PLATFORM=offscreen")
    endif()
    return()
endif()
