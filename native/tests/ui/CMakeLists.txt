cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NickelUiTests LANGUAGES CXX)
    include(CTest)
    if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    include("${CMAKE_CURRENT_LIST_DIR}/ConfigureMsvc.cmake")
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../ui" "${CMAKE_CURRENT_BINARY_DIR}/ui")
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
if(NOT NICKEL_UI_DOMAIN_INCLUDE_ROOT)
    get_filename_component(NICKEL_UI_DOMAIN_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/FixtureAssets.cmake")

function(nickel_add_ui_test target source)
    add_executable(${target} ${source} TestServices.hpp)
    set_target_properties(${target} PROPERTIES AUTOMOC ON)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_link_libraries(${target} PRIVATE nickel_ui Qt6::Test nickel_ui_fixture_assets ${ARGN})
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
    endif()
    if(TARGET Qt6::QOffscreenIntegrationPlugin)
        qt_import_plugins(${target} INCLUDE_BY_TYPE platforms Qt6::QOffscreenIntegrationPlugin)
    endif()
    if(WIN32 AND TARGET Qt6::QWindowsIntegrationPlugin)
        qt_import_plugins(${target} INCLUDE Qt6::QWindowsIntegrationPlugin)
    endif()
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES LABELS "ui;native;fixture" TIMEOUT 120)
endfunction()

nickel_add_ui_test(nickel_ui_tests ui_tests.cpp)
nickel_add_ui_test(nickel_ui_description_tests description_tests.cpp)
nickel_add_ui_test(nickel_ui_adapter_tests adapter_tests.cpp)
nickel_add_ui_test(nickel_ui_navigation_tests navigation_tests.cpp)
nickel_add_ui_test(nickel_ui_effects_tests effects_tests.cpp Qt6::Svg)
nickel_add_ui_test(nickel_ui_interaction_tests interaction_tests.cpp)
