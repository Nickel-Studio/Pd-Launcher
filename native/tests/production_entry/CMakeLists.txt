cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
project(palladium_production_entry_subset LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "The bounded entry subset must use strict Release objects")
endif()
if(NOT NICKEL_ENTRY_LINK_PORTAL OR NOT EXISTS "${NICKEL_ENTRY_LINK_PORTAL}")
    message(FATAL_ERROR "A builder-published same-generation root library portal is required; no second root build")
endif()
include("${NICKEL_ENTRY_LINK_PORTAL}")
if(NOT NICKEL_ENTRY_BUILD_INFO_DIRECTORY OR NOT EXISTS "${NICKEL_ENTRY_BUILD_INFO_DIRECTORY}/BuildInfo.hpp")
    message(FATAL_ERROR "The exact root-generated BuildInfo.hpp is required, not a test identity")
endif()
enable_testing()
include("${CMAKE_CURRENT_LIST_DIR}/production_tests.cmake")
