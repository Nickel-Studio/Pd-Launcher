    endif()
    target_include_directories(nickel_common BEFORE INTERFACE "${NICKEL_INSTANCES_IDENTITY_NATIVE_DIR}")
    if(NOT TARGET ZLIB::ZLIB)
        set(NICKEL_INSTANCES_ZLIB_SOURCE_DIR "" CACHE PATH "Actual local zlib source for standalone static PNG tests")
        if(NICKEL_INSTANCES_ZLIB_SOURCE_DIR)
            set(_zlib "${NICKEL_INSTANCES_ZLIB_SOURCE_DIR}")
            if(NOT EXISTS "${_zlib}/zlib.h" OR NOT EXISTS "${_zlib}/deflate.c")
                message(FATAL_ERROR "Standalone PNG tests require actual zlib sources")
            endif()
            if(EXISTS "${_zlib}/zconf.h")
                set(_zconf "${_zlib}/zconf.h")
            elseif(EXISTS "${_zlib}/zconf.h.included")
                set(_zconf "${_zlib}/zconf.h.included")
            else()
                message(FATAL_ERROR "Actual zlib source is missing its portable zconf header")
            endif()
            configure_file("${_zconf}" "${CMAKE_CURRENT_BINARY_DIR}/zlib-include/zconf.h" COPYONLY)
            add_library(nickel_instances_test_zlib STATIC
                "${_zlib}/adler32.c" "${_zlib}/compress.c" "${_zlib}/crc32.c" "${_zlib}/deflate.c"
                "${_zlib}/inflate.c" "${_zlib}/infback.c" "${_zlib}/inftrees.c" "${_zlib}/inffast.c"
                "${_zlib}/trees.c" "${_zlib}/uncompr.c" "${_zlib}/zutil.c")
            target_include_directories(nickel_instances_test_zlib PUBLIC "${_zlib}" "${CMAKE_CURRENT_BINARY_DIR}/zlib-include")
            add_library(ZLIB::ZLIB ALIAS nickel_instances_test_zlib)
        else()
            set(ZLIB_USE_STATIC_LIBS ON)
            find_package(ZLIB REQUIRED)
        endif()
    endif()
    set(NICKEL_IDENTITY_CONTROLLED_ONLY ON CACHE BOOL "Standalone tests use the actual identity/media core with controlled HTTP" FORCE)
    add_subdirectory("${NICKEL_INSTANCES_IDENTITY_NATIVE_DIR}/core/accounts" "${CMAKE_CURRENT_BINARY_DIR}/shared-identity")
endif()
