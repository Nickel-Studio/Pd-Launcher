cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(nickel_minecraft_domain_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    # Some installations lack English resources, so VSLANG alone cannot fix include tracking.
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file: ")
    set(CMAKE_CXX_COMPILER_LAUNCHER "${Python3_EXECUTABLE};${CMAKE_CURRENT_LIST_DIR}/../msvc-includes.py")
endif()
get_filename_component(NICKEL_WORKTREE "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
find_package(nlohmann_json 3.11.3 CONFIG REQUIRED)
add_library(nickel_common INTERFACE)
target_compile_features(nickel_common INTERFACE cxx_std_20)
target_include_directories(nickel_common INTERFACE "${NICKEL_WORKTREE}/native")
target_link_libraries(nickel_common INTERFACE nlohmann_json::nlohmann_json)
option(NICKEL_MINECRAFT_REQUIRE_NATIVE_SERVICES "Build actual foundation/transport production adapters" OFF)
if(NICKEL_MINECRAFT_REQUIRE_NATIVE_SERVICES)
    set(NICKEL_SHARED_NATIVE "${NICKEL_WORKTREE}/.snow/native-shared/native" CACHE PATH "Actual dependency snapshot include root")
    target_include_directories(nickel_common BEFORE INTERFACE "${NICKEL_SHARED_NATIVE}")
    # Parent scope keeps the imported target visible to the standalone link checks.
    find_package(LibArchive REQUIRED)
    if(WIN32 AND LibArchive_LIBRARY MATCHES "archive\\.lib$")
        set(ZLIB_USE_STATIC_LIBS ON)
        find_package(ZLIB REQUIRED)
        find_package(BZip2 REQUIRED)
        find_package(LibLZMA REQUIRED)
        find_package(zstd CONFIG REQUIRED)
        set_property(TARGET LibArchive::LibArchive APPEND PROPERTY INTERFACE_COMPILE_DEFINITIONS LIBARCHIVE_STATIC)
        set_property(TARGET LibArchive::LibArchive APPEND PROPERTY INTERFACE_LINK_LIBRARIES
            ZLIB::ZLIB BZip2::BZip2 LibLZMA::LibLZMA zstd::libzstd_static bcrypt xmllite uuid)
    endif()
    add_subdirectory("${NICKEL_SHARED_NATIVE}/core/config" "${CMAKE_CURRENT_BINARY_DIR}/foundation")
    add_subdirectory("${NICKEL_SHARED_NATIVE}/core/network" "${CMAKE_CURRENT_BINARY_DIR}/transport")
endif()
add_subdirectory("${NICKEL_WORKTREE}/native/core/minecraft" "${CMAKE_CURRENT_BINARY_DIR}/minecraft")
enable_testing()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/tests")
