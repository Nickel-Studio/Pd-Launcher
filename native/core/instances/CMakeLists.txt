cmake_minimum_required(VERSION 3.24)

if(NOT TARGET nickel_common OR NOT TARGET nickel_foundation OR NOT TARGET nickel_identity)
    message(FATAL_ERROR "Instances require canonical nickel_common, nickel_foundation and nickel_identity from the root build or standalone tests")
endif()
add_library(nickel_instances STATIC domain_io.cpp instances.cpp settings.cpp startup.cpp icons.cpp)
target_compile_features(nickel_instances PUBLIC cxx_std_20)
target_link_libraries(nickel_instances PUBLIC nickel_common nickel_foundation nickel_identity)
add_subdirectory(../diagnostics "${CMAKE_CURRENT_BINARY_DIR}/diagnostics")
add_subdirectory(../packs "${CMAKE_CURRENT_BINARY_DIR}/packs")
if(WIN32)
    target_link_libraries(nickel_instances PRIVATE bcrypt)
else()
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    target_link_libraries(nickel_instances PRIVATE OpenSSL::Crypto)
endif()
if(MSVC)
    target_compile_options(nickel_instances PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(nickel_instances PRIVATE -Wall -Wextra -Wpedantic)
endif()
