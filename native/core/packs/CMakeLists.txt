cmake_minimum_required(VERSION 3.24)
if(NOT TARGET nickel_instances)
    message(FATAL_ERROR "Pack sources belong to the nickel_instances target")
endif()
if(NICKEL_INSTANCES_YAML_TARGET AND TARGET "${NICKEL_INSTANCES_YAML_TARGET}")
    target_link_libraries(nickel_instances PRIVATE "${NICKEL_INSTANCES_YAML_TARGET}")
else()
    find_path(NICKEL_INSTANCES_YAML_INCLUDE_DIR yaml.h REQUIRED)
    find_library(NICKEL_INSTANCES_YAML_LIBRARY NAMES yaml yaml_static REQUIRED)
    target_include_directories(nickel_instances PRIVATE "${NICKEL_INSTANCES_YAML_INCLUDE_DIR}")
    target_link_libraries(nickel_instances PRIVATE "${NICKEL_INSTANCES_YAML_LIBRARY}")
    if(WIN32)
        target_compile_definitions(nickel_instances PRIVATE YAML_DECLARE_STATIC)
    endif()
endif()
get_filename_component(_nickel_instances_test_root "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/instances" ABSOLUTE)
if("${CMAKE_SOURCE_DIR}" STREQUAL "${_nickel_instances_test_root}")
    find_path(NICKEL_INSTANCES_TOML_INCLUDE_DIR NAMES toml++/toml.hpp REQUIRED)
    target_include_directories(nickel_instances PRIVATE "${NICKEL_INSTANCES_TOML_INCLUDE_DIR}")
else()
    if(NOT TARGET tomlplusplus::tomlplusplus)
        find_package(tomlplusplus 3.4 REQUIRED CONFIG)
    endif()
    target_link_libraries(nickel_instances PRIVATE tomlplusplus::tomlplusplus)
endif()
target_sources(nickel_instances PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/internal.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/hash.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/policy.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/packs.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/export.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/portable.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/native_contracts.cpp")

if("${CMAKE_SOURCE_DIR}" STREQUAL "${_nickel_instances_test_root}")
    # Standalone tests compile real bindings, but never fabricate foreign link symbols.
    add_library(nickel_instances_native_bindings_compile OBJECT native_archive.cpp native_services.cpp)
    target_compile_features(nickel_instances_native_bindings_compile PRIVATE cxx_std_20)
    target_link_libraries(nickel_instances_native_bindings_compile PRIVATE nickel_instances)
    if(MSVC)
        target_compile_options(nickel_instances_native_bindings_compile PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(nickel_instances_native_bindings_compile PRIVATE -Wall -Wextra -Wpedantic)
    endif()
else()
    target_sources(nickel_instances PRIVATE native_archive.cpp native_services.cpp)
    target_link_libraries(nickel_instances PRIVATE nickel_transport nickel_minecraft nickel_community)
    function(nickel_instances_check_native_targets)
        foreach(_target IN ITEMS nickel_transport nickel_minecraft nickel_community)
            if(NOT TARGET "${_target}")
                message(FATAL_ERROR "Production instances require the actual ${_target} target; add its owner sources to the root build")
            endif()
        endforeach()
    endfunction()
    cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL nickel_instances_check_native_targets)
endif()

# Production ZIP/TAR stays transport-owned. This real ZIP backend is opt-in test infrastructure only.
if(NICKEL_INSTANCES_ZIP_TEST_BACKEND)
    if(NOT NICKEL_INSTANCES_MINIZ_SOURCE_DIR)
        message(FATAL_ERROR "Real ZIP test backend requires a supplied miniz 3.1.0 source directory")
    endif()
    enable_language(C)
    set(_miniz "${NICKEL_INSTANCES_MINIZ_SOURCE_DIR}")
    add_library(nickel_instances_test_miniz STATIC
        "${_miniz}/miniz.c" "${_miniz}/miniz_zip.c" "${_miniz}/miniz_tdef.c" "${_miniz}/miniz_tinfl.c")
    target_include_directories(nickel_instances_test_miniz PUBLIC "${_miniz}")
    target_compile_definitions(nickel_instances_test_miniz PUBLIC MINIZ_NO_ZLIB_COMPATIBLE_NAMES MINIZ_NO_STDIO)
    add_library(nickel_instances_zip_test_backend STATIC archive_bridge.cpp)
    target_compile_features(nickel_instances_zip_test_backend PRIVATE cxx_std_20)
    target_link_libraries(nickel_instances_zip_test_backend PUBLIC nickel_instances PRIVATE nickel_instances_test_miniz)
    if(MSVC)
        target_compile_options(nickel_instances_zip_test_backend PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(nickel_instances_zip_test_backend PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
