cmake_minimum_required(VERSION 3.25)
project(NickelIndependentPortability LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

set(NICKEL_NATIVE_ROOT "" CACHE PATH "Exact read-only owner snapshot native directory")
if(NOT EXISTS "${NICKEL_NATIVE_ROOT}/core/config/CMakeLists.txt")
    message(FATAL_ERROR "NICKEL_NATIVE_ROOT must identify the actual native owner snapshot")
endif()
find_package(nlohmann_json 3.11 CONFIG REQUIRED)
find_package(Threads REQUIRED)

# The owner common target assumes the product root. This test target exposes the
# same real headers and JSON library, without injecting a substitute implementation.
add_library(nickel_common INTERFACE)
target_compile_features(nickel_common INTERFACE cxx_std_20)
target_include_directories(nickel_common INTERFACE "${NICKEL_NATIVE_ROOT}")
target_link_libraries(nickel_common INTERFACE nlohmann_json::nlohmann_json)
add_subdirectory("${NICKEL_NATIVE_ROOT}/core/config" "${CMAKE_CURRENT_BINARY_DIR}/foundation")

add_executable(nickel_portability_probe portability_probe.cpp)
target_link_libraries(nickel_portability_probe PRIVATE nickel_foundation nickel_common Threads::Threads)
if(MSVC)
    enable_language(RC)
    target_sources(nickel_portability_probe PRIVATE probe.rc)
    target_compile_options(nickel_portability_probe PRIVATE /W4 /utf-8 /permissive-)
    target_link_options(nickel_portability_probe PRIVATE /MANIFEST:NO)
else()
    target_compile_options(nickel_portability_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(NICKEL_PORTABILITY_ALL_CORE "Compile and whole-archive link actual std-core owner targets; Qt adapter is excluded explicitly" OFF)
if(NICKEL_PORTABILITY_ALL_CORE)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto SSL)
    find_package(tomlplusplus CONFIG REQUIRED)
    if(NOT TARGET gumbo::gumbo)
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(GUMBO REQUIRED IMPORTED_TARGET gumbo)
        add_library(gumbo::gumbo ALIAS PkgConfig::GUMBO)
    endif()
    set(NICKEL_COMMUNITY_QT_ADAPTER OFF CACHE BOOL "Qt UI adapter is not part of this std-core fixture" FORCE)
    set(NICKEL_IDENTITY_CONTROLLED_ONLY OFF CACHE BOOL "Never replace real transport/foundation" FORCE)
    set(NICKEL_MINECRAFT_REQUIRE_NATIVE_SERVICES ON)
    add_subdirectory("${NICKEL_NATIVE_ROOT}/core/network" "${CMAKE_CURRENT_BINARY_DIR}/transport")
    add_subdirectory("${NICKEL_NATIVE_ROOT}/core/accounts" "${CMAKE_CURRENT_BINARY_DIR}/identity")
    add_subdirectory("${NICKEL_NATIVE_ROOT}/core/instances" "${CMAKE_CURRENT_BINARY_DIR}/instances")
    add_subdirectory("${NICKEL_NATIVE_ROOT}/core/resources" "${CMAKE_CURRENT_BINARY_DIR}/community")
    add_subdirectory("${NICKEL_NATIVE_ROOT}/core/minecraft" "${CMAKE_CURRENT_BINARY_DIR}/minecraft")
    add_executable(nickel_portability_core_link core_link_probe.cpp)
    foreach(_core nickel_foundation nickel_transport nickel_identity nickel_instances nickel_community nickel_minecraft)
        target_link_libraries(nickel_portability_core_link PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,${_core}>")
        set_property(TARGET nickel_portability_core_link PROPERTY "LINK_LIBRARY_OVERRIDE_${_core}" WHOLE_ARCHIVE)
    endforeach()
    if(MSVC)
        target_compile_options(nickel_portability_core_link PRIVATE /W4 /utf-8 /permissive-)
    else()
        target_compile_options(nickel_portability_core_link PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    message(STATUS "Scope: real std-core owner targets, full symbol closure; Qt adapter/app/UI and exact release dependency pins NOT tested")
endif()

enable_testing()
add_test(NAME portability.real_foundation COMMAND nickel_portability_probe
    --fixture-root "${CMAKE_CURRENT_BINARY_DIR}/fixture-root")
set_tests_properties(portability.real_foundation PROPERTIES TIMEOUT 90)
