cmake_minimum_required(VERSION 3.25)
project(PdCMakeGenerationRegression LANGUAGES NONE)

set(PD_REGRESSION_MODE "raw" CACHE STRING "raw or builder")
set(PD_DOMAIN_MODULE "" CACHE FILEPATH "Original builder-owned NickelDomains module")
set_property(CACHE PD_REGRESSION_MODE PROPERTY STRINGS raw builder)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This regression requires the real Ninja generator")
endif()

if(PD_REGRESSION_MODE STREQUAL "builder")
    if(NOT IS_ABSOLUTE "${PD_DOMAIN_MODULE}" OR NOT EXISTS "${PD_DOMAIN_MODULE}")
        message(FATAL_ERROR "builder mode must include an existing original module by absolute path")
    endif()
    include("${PD_DOMAIN_MODULE}")
elseif(NOT PD_REGRESSION_MODE STREQUAL "raw")
    message(FATAL_ERROR "Unknown regression mode: ${PD_REGRESSION_MODE}")
endif()

add_subdirectory(native/app)
if(PD_REGRESSION_MODE STREQUAL "builder")
    nickel_add_domain_tests(brand brand)
else()
    add_subdirectory(native/tests/brand)
endif()
