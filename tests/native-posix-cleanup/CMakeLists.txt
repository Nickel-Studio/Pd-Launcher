cmake_minimum_required(VERSION 3.25)
project(NickelPosixCleanupAcceptance LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This acceptance must execute the real Linux production branch")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ACCEPTANCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(NICKEL_POSIX_SOURCE_ROOT "" CACHE PATH "Explicit completed 24-file production input root")
set(NICKEL_POSIX_MANIFEST_SHA256 "" CACHE STRING "Pinned receiver manifest SHA-256")
if(NOT NICKEL_POSIX_SOURCE_ROOT OR NOT NICKEL_POSIX_MANIFEST_SHA256)
    message(FATAL_ERROR "An explicit production root and pinned manifest SHA are required")
endif()
file(REAL_PATH "${NICKEL_POSIX_SOURCE_ROOT}" SHARED_ROOT)
string(FIND "${SHARED_ROOT}/" "${ACCEPTANCE_ROOT}/.snow/" owner_prefix)
if(NOT owner_prefix EQUAL 0 OR EXISTS "${SHARED_ROOT}/.publisher.lock")
    message(FATAL_ERROR "Production root must be a completed owner-local snapshot")
endif()
file(READ "${SHARED_ROOT}/publication-complete.json" completion)
string(JSON complete GET "${completion}" publicationComplete)
string(JSON expected_manifest ERROR_VARIABLE key_error GET "${completion}" receiverManifestSha256)
if(key_error)
    string(JSON expected_manifest GET "${completion}" snapshotFileSha256)
endif()
file(SHA256 "${SHARED_ROOT}/snapshot.json" actual_manifest)
if(NOT complete OR NOT expected_manifest STREQUAL actual_manifest OR NOT NICKEL_POSIX_MANIFEST_SHA256 STREQUAL actual_manifest)
    message(FATAL_ERROR "Completed pinned snapshot SHA is required: ${SHARED_ROOT}")
endif()
file(READ "${SHARED_ROOT}/snapshot.json" snapshot)
string(JSON count LENGTH "${snapshot}" files)
if(NOT count EQUAL 24)
    message(FATAL_ERROR "The selected production root must contain the complete 24-file closure")
endif()
math(EXPR last "${count} - 1")
foreach(index RANGE 0 ${last})
    string(JSON input MEMBER "${snapshot}" files ${index})
    string(JSON expected GET "${snapshot}" files "${input}" sha256)
    string(JSON expected_bytes GET "${snapshot}" files "${input}" bytes)
    file(SHA256 "${SHARED_ROOT}/${input}" actual)
    file(SIZE "${SHARED_ROOT}/${input}" actual_bytes)
    if(NOT actual STREQUAL expected OR NOT actual_bytes EQUAL expected_bytes)
        message(FATAL_ERROR "Frozen input mismatch: ${SHARED_ROOT}/${input}")
    endif()
endforeach()
message(STATUS "Completed POSIX snapshot matched ${count} inputs; production_only=${SHARED_ROOT}; manifest=${actual_manifest}")
find_package(nlohmann_json 3.11 REQUIRED CONFIG)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
set(PRODUCTION "${SHARED_ROOT}/native")
add_library(acceptance_base OBJECT
    "${PRODUCTION}/core/diagnostics/junk.cpp"
    "${PRODUCTION}/core/config/atomic_file.cpp"
    "${PRODUCTION}/core/config/path_guard.cpp"
    "${PRODUCTION}/core/config/detail.cpp"
    "${PRODUCTION}/core/config/crypto.cpp"
    "${PRODUCTION}/core/network/control.cpp"
    "${PRODUCTION}/core/network/hash.cpp"
)
target_include_directories(acceptance_base PUBLIC "${PRODUCTION}")
target_link_libraries(acceptance_base PUBLIC nlohmann_json::nlohmann_json OpenSSL::Crypto Threads::Threads)
target_compile_options(acceptance_base PRIVATE -Wall -Wextra -Wpedantic)

add_executable(posix_cleanup_plain acceptance.cpp "${PRODUCTION}/core/tasks/secure_fs.cpp")
target_link_libraries(posix_cleanup_plain PRIVATE acceptance_base)
target_compile_options(posix_cleanup_plain PRIVATE -Wall -Wextra -Wpedantic)

add_executable(posix_cleanup_controlled acceptance.cpp "${PRODUCTION}/core/tasks/secure_fs.cpp")
target_link_libraries(posix_cleanup_controlled PRIVATE acceptance_base)
target_compile_definitions(posix_cleanup_controlled PRIVATE NICKEL_TRANSPORT_TEST_HOOKS POSIX_ACCEPTANCE_OBSERVERS)
target_compile_options(posix_cleanup_controlled PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(posix_cleanup_controlled PRIVATE -Wl,--wrap=read -Wl,--wrap=syscall -Wl,--wrap=unlinkat)

include(CTest)
add_test(NAME posix_cleanup_plain COMMAND posix_cleanup_plain)
add_test(NAME posix_cleanup_controlled COMMAND posix_cleanup_controlled)
set_tests_properties(posix_cleanup_plain posix_cleanup_controlled PROPERTIES RUN_SERIAL TRUE TIMEOUT 90)
