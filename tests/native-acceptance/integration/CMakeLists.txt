cmake_minimum_required(VERSION 3.28)
cmake_policy(SET CMP0091 NEW)
if(POLICY CMP0176)
    cmake_policy(SET CMP0176 NEW)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(NickelIndependentProductionIntegration LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set_property(GLOBAL PROPERTY JOB_POOLS acceptance_compile=1 acceptance_link=1)
set(CMAKE_JOB_POOL_COMPILE acceptance_compile)
set(CMAKE_JOB_POOL_LINK acceptance_link)
enable_testing()
get_filename_component(_workspace "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(NICKEL_SOURCE_ROOT "${_workspace}/.snow/native-shared" CACHE PATH "Exact frozen production snapshot root")
file(REAL_PATH "${NICKEL_SOURCE_ROOT}" NICKEL_SOURCE_ROOT)
set(_native "${NICKEL_SOURCE_ROOT}/native")
foreach(_scope core/common core/config core/network core/accounts core/instances core/resources core/minecraft)
    if(NOT EXISTS "${_native}/${_scope}/CMakeLists.txt")
        message(FATAL_ERROR "Missing actual production target at ${_native}/${_scope}")
    endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(MSVC)
    file(REAL_PATH "${CMAKE_CURRENT_BINARY_DIR}" _probe_directory)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${_native}/tests/foundation/msvc_showincludes.py"
        "${CMAKE_CXX_COMPILER}" "${_probe_directory}"
        RESULT_VARIABLE _probe_result OUTPUT_VARIABLE _include_prefix ERROR_VARIABLE _probe_error ENCODING UTF-8)
    if(NOT _probe_result EQUAL 0)
        message(FATAL_ERROR "Actual MSVC include-prefix probe failed: ${_probe_error}")
    endif()
    string(REGEX REPLACE "[\r\n]+$" "" _include_prefix "${_include_prefix}")
    set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${_include_prefix}")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${_include_prefix}")
endif()
include("${NICKEL_SOURCE_ROOT}/cmake/NickelDependencies.cmake")
set(NICKEL_INSTANCES_YAML_TARGET yaml::yaml)
set(NICKEL_IDENTITY_CONTROLLED_ONLY OFF CACHE BOOL "Production owner adapters are mandatory" FORCE)
set(NICKEL_MINECRAFT_REQUIRE_NATIVE_SERVICES ON CACHE BOOL "Actual platform, HTTP and archive adapters are mandatory" FORCE)
set(NICKEL_COMMUNITY_QT_ADAPTER OFF CACHE BOOL "Headless core verification; Qt/UI remains a separate gate" FORCE)
add_subdirectory("${_native}/core/common" "${CMAKE_CURRENT_BINARY_DIR}/common")
add_subdirectory("${_native}/core/config" "${CMAKE_CURRENT_BINARY_DIR}/foundation")
add_subdirectory("${_native}/core/network" "${CMAKE_CURRENT_BINARY_DIR}/transport")
add_subdirectory("${_native}/core/accounts" "${CMAKE_CURRENT_BINARY_DIR}/identity")
add_subdirectory("${_native}/core/minecraft" "${CMAKE_CURRENT_BINARY_DIR}/minecraft")
add_subdirectory("${_native}/core/instances" "${CMAKE_CURRENT_BINARY_DIR}/instances")
add_subdirectory("${_native}/core/resources" "${CMAKE_CURRENT_BINARY_DIR}/community")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/independent")
file(REAL_PATH "${CMAKE_CURRENT_BINARY_DIR}" _physical_build_directory)
foreach(_domain foundation transport identity instances minecraft)
    if(_domain STREQUAL "transport")
        # Fixture paths must not name the relocated cache's junction.
        set(_test_directory "${_physical_build_directory}/tests/${_domain}")
    else()
        set(_test_directory "${CMAKE_CURRENT_BINARY_DIR}/tests/${_domain}")
    endif()
    add_subdirectory("${_native}/tests/${_domain}" "${_test_directory}")
endforeach()

# Use the unchanged owner suite and the actual aggregate; never a core-only substitute.
add_executable(nickel_acceptance_community_contracts
    "${_native}/tests/community/community_tests.cpp" "${_native}/tests/community/contracts_tests.cpp")
target_link_libraries(nickel_acceptance_community_contracts PRIVATE Nickel::Community)
if(MSVC)
    target_compile_options(nickel_acceptance_community_contracts PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()
add_test(NAME native_community_actual_target_contracts COMMAND nickel_acceptance_community_contracts)
set_tests_properties(native_community_actual_target_contracts PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)

set(NICKEL_ACCEPTANCE_SCRATCH "" CACHE PATH "Private physical fixture workdir; production path guards remain unchanged")
if(NICKEL_ACCEPTANCE_SCRATCH)
    file(REAL_PATH "${NICKEL_ACCEPTANCE_SCRATCH}" _scratch)
    if(NOT IS_DIRECTORY "${_scratch}")
        message(FATAL_ERROR "Create a private physical fixture workdir first")
    endif()
    foreach(_domain foundation transport instances minecraft)
        get_property(_tests DIRECTORY "${_native}/tests/${_domain}" PROPERTY TESTS)
        if(_tests)
            set_tests_properties(${_tests} DIRECTORY "${_native}/tests/${_domain}" PROPERTIES WORKING_DIRECTORY "${_scratch}")
        endif()
    endforeach()
    set_tests_properties(native_community_actual_target_contracts PROPERTIES WORKING_DIRECTORY "${_scratch}")
endif()
