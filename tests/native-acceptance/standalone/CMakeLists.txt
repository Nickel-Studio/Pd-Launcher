cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(NickelIndependentAcceptance LANGUAGES CXX)
enable_testing()
get_filename_component(_default_source "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(NICKEL_SOURCE_ROOT "${_default_source}" CACHE PATH "Exact production source root under test")
find_package(nlohmann_json 3.11 CONFIG REQUIRED)
add_library(nickel_common INTERFACE)
target_compile_features(nickel_common INTERFACE cxx_std_20)
target_include_directories(nickel_common INTERFACE "${NICKEL_SOURCE_ROOT}/native")
target_link_libraries(nickel_common INTERFACE nlohmann_json::nlohmann_json)
add_subdirectory("${NICKEL_SOURCE_ROOT}/native/core/config" "${CMAKE_CURRENT_BINARY_DIR}/foundation")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/independent")
option(NICKEL_VERIFY_OWNER_TESTS "Run the supplied foundation compatibility suite too" ON)
if(NICKEL_VERIFY_OWNER_TESTS)
    add_subdirectory("${NICKEL_SOURCE_ROOT}/native/tests/foundation" "${CMAKE_CURRENT_BINARY_DIR}/owner-tests")
endif()
