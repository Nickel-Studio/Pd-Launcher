if(NOT TARGET nickel_foundation)
    message(FATAL_ERROR "Independent acceptance requires the real nickel_foundation target")
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_executable(nickel_foundation_golden foundation_golden.cpp)
target_link_libraries(nickel_foundation_golden PRIVATE nickel_foundation nickel_common)
target_compile_features(nickel_foundation_golden PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(nickel_foundation_golden PRIVATE /W4 /utf-8)
else()
    target_compile_options(nickel_foundation_golden PRIVATE -Wall -Wextra -Wpedantic)
endif()
set(_golden_dir "${CMAKE_CURRENT_BINARY_DIR}/golden-fixtures")
add_test(NAME native_golden_generate COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/generate_fixtures.py" --output "${_golden_dir}")
set_tests_properties(native_golden_generate PROPERTIES FIXTURES_SETUP native_golden TIMEOUT 30)
add_test(NAME native_foundation_golden COMMAND nickel_foundation_golden "${_golden_dir}")
set_tests_properties(native_foundation_golden PROPERTIES FIXTURES_REQUIRED native_golden TIMEOUT 120)
add_test(NAME native_acceptance_tools COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_tools.py")
set_tests_properties(native_acceptance_tools PROPERTIES TIMEOUT 60)
add_test(NAME native_acceptance_record_tools COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_record_ctest.py")
set_tests_properties(native_acceptance_record_tools PROPERTIES TIMEOUT 60)
