cmake_minimum_required(VERSION 3.25)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(PalladiumCleanupRegression LANGUAGES CXX)
    include(CTest)
endif()

get_filename_component(NICKEL_CLEANUP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(NICKEL_CLEANUP_SHARED_ROOT "" CACHE PATH "Completed immutable production dependency snapshot")
set(NICKEL_CLEANUP_MANIFEST_SHA256 "" CACHE STRING "Expected SHA256 of the completed dependency manifest")
set(NICKEL_CLEANUP_FIXTURE_ROOT "${CMAKE_CURRENT_BINARY_DIR}/fixtures" CACHE PATH "Owned isolated cleanup fixtures")
set(cleanup_owned_target FALSE)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(NOT NICKEL_CLEANUP_SHARED_ROOT OR NOT NICKEL_CLEANUP_MANIFEST_SHA256)
        message(FATAL_ERROR "Supply an exact completed dependency snapshot and manifest SHA256; no old native fallback")
    endif()
    set(cleanup_manifest_path "${NICKEL_CLEANUP_SHARED_ROOT}/snapshot.json")
    set(cleanup_marker_path "${NICKEL_CLEANUP_SHARED_ROOT}/snapshot-complete.json")
    if(NOT EXISTS "${cleanup_manifest_path}" OR NOT EXISTS "${cleanup_marker_path}")
        message(FATAL_ERROR "Cleanup dependency publication is incomplete")
    endif()
    file(SHA256 "${cleanup_manifest_path}" cleanup_manifest_hash)
    file(READ "${cleanup_manifest_path}" cleanup_manifest)
    file(READ "${cleanup_marker_path}" cleanup_marker)
    string(JSON cleanup_marker_hash GET "${cleanup_marker}" manifestSha256)
    string(JSON cleanup_complete GET "${cleanup_marker}" complete)
    if(NOT cleanup_complete OR NOT cleanup_manifest_hash STREQUAL NICKEL_CLEANUP_MANIFEST_SHA256 OR
       NOT cleanup_marker_hash STREQUAL cleanup_manifest_hash)
        message(FATAL_ERROR "Cleanup dependency completion marker or manifest hash mismatch")
    endif()
    string(JSON cleanup_file_count LENGTH "${cleanup_manifest}" files)
    if(cleanup_file_count LESS 1)
        message(FATAL_ERROR "Cleanup dependency publication is empty")
    endif()
    math(EXPR cleanup_file_last "${cleanup_file_count} - 1")
    foreach(cleanup_file_index RANGE 0 ${cleanup_file_last})
        string(JSON cleanup_file MEMBER "${cleanup_manifest}" files ${cleanup_file_index})
        if(IS_ABSOLUTE "${cleanup_file}" OR cleanup_file MATCHES "(^|/)\\.\\.(/|$)" OR cleanup_file MATCHES "\\\\")
            message(FATAL_ERROR "Invalid cleanup dependency path: ${cleanup_file}")
        endif()
        string(JSON cleanup_expected GET "${cleanup_manifest}" files "${cleanup_file}" sha256)
        string(JSON cleanup_expected_bytes GET "${cleanup_manifest}" files "${cleanup_file}" bytes)
        set(cleanup_path "${NICKEL_CLEANUP_SHARED_ROOT}/${cleanup_file}")
        if(NOT EXISTS "${cleanup_path}" OR IS_SYMLINK "${cleanup_path}")
            message(FATAL_ERROR "Missing regular cleanup dependency: ${cleanup_file}")
        endif()
        file(SHA256 "${cleanup_path}" cleanup_actual)
        file(SIZE "${cleanup_path}" cleanup_actual_bytes)
        if(NOT cleanup_actual STREQUAL cleanup_expected OR NOT cleanup_actual_bytes EQUAL cleanup_expected_bytes)
            message(FATAL_ERROR "Cleanup dependency SHA256 or byte mismatch: ${cleanup_file}")
        endif()
    endforeach()
    message(STATUS "Palladium cleanup provenance matched ${cleanup_file_count} immutable Git inputs")

    find_package(nlohmann_json 3.11 REQUIRED CONFIG)
    find_package(Threads REQUIRED)
    set(cleanup_dependencies "${NICKEL_CLEANUP_SHARED_ROOT}/native")
    add_library(nickel_cleanup_safety STATIC
        "${cleanup_dependencies}/core/config/atomic_file.cpp"
        "${cleanup_dependencies}/core/config/path_guard.cpp"
        "${cleanup_dependencies}/core/config/detail.cpp"
        "${cleanup_dependencies}/core/config/crypto.cpp"
        "${cleanup_dependencies}/core/tasks/secure_fs.cpp"
        "${cleanup_dependencies}/core/network/hash.cpp"
        "${cleanup_dependencies}/core/network/control.cpp"
    )
    target_include_directories(nickel_cleanup_safety PUBLIC "${cleanup_dependencies}")
    target_link_libraries(nickel_cleanup_safety PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
    target_compile_features(nickel_cleanup_safety PUBLIC cxx_std_20)
    target_compile_definitions(nickel_cleanup_safety PUBLIC NICKEL_TRANSPORT_TEST_HOOKS)
    if(WIN32)
        target_link_libraries(nickel_cleanup_safety PUBLIC bcrypt advapi32)
    else()
        find_package(OpenSSL REQUIRED)
        target_link_libraries(nickel_cleanup_safety PUBLIC OpenSSL::Crypto)
    endif()

    add_library(nickel_cleanup STATIC "${NICKEL_CLEANUP_ROOT}/native/core/diagnostics/junk.cpp")
    target_include_directories(nickel_cleanup BEFORE PUBLIC "${cleanup_dependencies}" "${NICKEL_CLEANUP_ROOT}/native")
    target_link_libraries(nickel_cleanup PUBLIC nickel_cleanup_safety)
    target_compile_features(nickel_cleanup PUBLIC cxx_std_20)
    set(cleanup_owned_target TRUE)
elseif(NOT TARGET nickel_cleanup)
    if(NOT TARGET nickel_instances)
        message(FATAL_ERROR "Integrated cleanup regression requires the production nickel_instances target")
    endif()
    get_target_property(cleanup_instance_sources nickel_instances SOURCES)
    set(cleanup_junk_source_count 0)
    foreach(cleanup_source IN LISTS cleanup_instance_sources)
        if(cleanup_source MATCHES "(^|[/\\\\])junk\\.cpp$")
            math(EXPR cleanup_junk_source_count "${cleanup_junk_source_count} + 1")
        endif()
    endforeach()
    if(NOT cleanup_junk_source_count EQUAL 1)
        message(FATAL_ERROR "Integrated cleanup must link exactly one production junk.cpp, found ${cleanup_junk_source_count}")
    endif()
    add_library(nickel_cleanup INTERFACE)
    target_link_libraries(nickel_cleanup INTERFACE nickel_instances)
endif()

add_executable(nickel_cleanup_tests cleanup_tests.cpp)
set_target_properties(nickel_cleanup_tests PROPERTIES OUTPUT_NAME palladium_cleanup_tests)
target_link_libraries(nickel_cleanup_tests PRIVATE nickel_cleanup)
target_compile_features(nickel_cleanup_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(nickel_cleanup_tests PRIVATE /W4 /WX /utf-8)
    if(cleanup_owned_target)
        target_compile_options(nickel_cleanup PRIVATE /W4 /WX /utf-8)
        target_compile_options(nickel_cleanup_safety PRIVATE /W4 /utf-8)
    endif()
else()
    target_compile_options(nickel_cleanup_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(cleanup_owned_target)
        target_compile_options(nickel_cleanup PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()

file(MAKE_DIRECTORY "${NICKEL_CLEANUP_FIXTURE_ROOT}")
add_test(NAME palladium_cleanup_adversarial COMMAND nickel_cleanup_tests "${NICKEL_CLEANUP_FIXTURE_ROOT}")
set_tests_properties(palladium_cleanup_adversarial PROPERTIES RUN_SERIAL TRUE TIMEOUT 120
    ENVIRONMENT "TEMP=${NICKEL_CLEANUP_FIXTURE_ROOT};TMP=${NICKEL_CLEANUP_FIXTURE_ROOT}")
