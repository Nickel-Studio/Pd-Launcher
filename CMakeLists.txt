cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native/metadata.json" _nickel_metadata)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/native/metadata.json")
string(JSON _nickel_version GET "${_nickel_metadata}" version)
project(Nickel VERSION "${_nickel_version}" LANGUAGES C CXX)
include(cmake/NickelBuildEnvironment.cmake)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_DISABLE_PRECOMPILE_HEADERS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_property(GLOBAL PROPERTY JOB_POOLS nickel_compile=1 nickel_link=1)
set(CMAKE_JOB_POOL_COMPILE nickel_compile)
set(CMAKE_JOB_POOL_LINK nickel_link)
option(NICKEL_BUILD_UI "Build the complete native Widgets application" ON)
option(NICKEL_BUILD_TESTS "Build real domain and independent acceptance tests" ON)
option(NICKEL_ENABLE_IPO "Enable IPO in complete release builds" OFF)
option(NICKEL_TEST_TRANSPORT_FAULTS "Enable private real transport checkpoints for the complete root test graph" ${NICKEL_BUILD_TESTS})
if(NICKEL_TEST_TRANSPORT_FAULTS AND NOT NICKEL_BUILD_TESTS)
    message(FATAL_ERROR "Transport fault checkpoints are restricted to an enabled root test graph")
endif()
if(NICKEL_BUILD_UI AND NICKEL_BUILD_TESTS AND NOT NICKEL_TEST_TRANSPORT_FAULTS)
    message(FATAL_ERROR "The complete application resource matrix requires actual transport fault checkpoints")
endif()
if(NICKEL_BUILD_TESTS OR NICKEL_TEST_TRANSPORT_FAULTS)
    # Fail before any subdirectory copies an instrumented application or dependency artifact.
    install(CODE [[message(FATAL_ERROR
        "Palladium candidate installation requires a separate tests-OFF/faults-OFF production build")]] ALL_COMPONENTS)
endif()
if(APPLE)
    # Apple's libc++ hides the C++20 stop-token family behind
    # _LIBCPP_HAS_NO_EXPERIMENTAL_STOP_TOKEN (verified on AppleClang 17 /
    # MacOSX15.5.sdk in CI run 45160199828) even under -std=c++20. Force-include
    # the stop-token compatibility layer into every C++ translation unit so the
    # family is re-enabled before <stop_token> or <thread> is parsed anywhere in
    # the project. This statement precedes every include()/add_subdirectory()
    # in this file so every target in the graph inherits the directory
    # COMPILE_OPTIONS property, and the path is shell-quoted inside SHELL: so
    # the UNIX_COMMAND-mode splitting keeps a path with spaces intact.
    # Windows/Linux keep the standard headers untouched.
    add_compile_options("SHELL:$<$<COMPILE_LANGUAGE:CXX>:-include \"${CMAKE_CURRENT_SOURCE_DIR}/native/core/common/stop_compat.hpp\">")
endif()

include(cmake/NickelDependencies.cmake)
include(cmake/NickelDomains.cmake)
add_subdirectory(native/core/common)
if(NICKEL_SHARED_NATIVE_ROOT)
    target_include_directories(nickel_common BEFORE INTERFACE "${NICKEL_SHARED_NATIVE_ROOT}")
endif()
if(MSVC)
    add_compile_options(/utf-8 /EHsc /permissive-)
    add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
endif()
set(NICKEL_INSTANCES_YAML_TARGET yaml::yaml)
set(NICKEL_MINECRAFT_REQUIRE_NATIVE_SERVICES ON)
set(NICKEL_COMMUNITY_QT_ADAPTER ${NICKEL_BUILD_UI} CACHE BOOL "Build the canonical Qt document adapter with the application" FORCE)
if(NICKEL_BUILD_UI)
    find_package(Qt6 6.8.3 EXACT CONFIG REQUIRED COMPONENTS Core Gui Widgets Svg)
    include(cmake/NickelQtDependencies.cmake)
    nickel_configure_qt_dependencies()
endif()
nickel_add_domain(core/config foundation)
nickel_add_domain(core/network transport)
if(NICKEL_TEST_TRANSPORT_FAULTS)
    target_compile_definitions(nickel_transport PRIVATE NICKEL_TRANSPORT_TEST_HOOKS=1)
endif()
set_property(TARGET nickel_transport PROPERTY NICKEL_TRANSPORT_FAULTS_ENABLED "${NICKEL_TEST_TRANSPORT_FAULTS}")
nickel_add_domain(core/accounts identity)
nickel_add_domain(core/instances instances)
nickel_domain_path(_nickel_diagnostics_source core/diagnostics)
foreach(_source junk.hpp junk.cpp)
    if(NOT EXISTS "${_nickel_diagnostics_source}/${_source}")
        message(FATAL_ERROR "Missing canonical cleanup source: ${_nickel_diagnostics_source}/${_source}")
    endif()
endforeach()
get_target_property(_nickel_instance_sources nickel_instances SOURCES)
if(NOT "${_nickel_diagnostics_source}/junk.cpp" IN_LIST _nickel_instance_sources AND NOT "junk.cpp" IN_LIST _nickel_instance_sources)
    target_sources(nickel_instances PRIVATE
        "${_nickel_diagnostics_source}/junk.hpp"
        "${_nickel_diagnostics_source}/junk.cpp")
endif()
nickel_add_domain(core/resources community)
nickel_add_domain(core/minecraft minecraft)
if(NICKEL_ENABLE_IPO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT _nickel_ipo_supported OUTPUT _nickel_ipo_error)
    if(NOT _nickel_ipo_supported)
        message(FATAL_ERROR "Requested IPO is unsupported: ${_nickel_ipo_error}")
    endif()
    foreach(_target nickel_foundation nickel_transport nickel_identity nickel_instances nickel_community nickel_minecraft)
        set_property(TARGET "${_target}" PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    endforeach()
endif()
if(NICKEL_BUILD_TESTS)
    include(CTest)
    enable_testing()
    foreach(_domain foundation transport identity instances community minecraft)
        nickel_add_domain_tests("${_domain}" "${_domain}")
    endforeach()
    add_subdirectory(tests/native-acceptance)
endif()
if(NICKEL_BUILD_UI)
    if(NOT TARGET Nickel::CommunityQt)
        message(FATAL_ERROR "The UI requires the canonical production CommunityQt document adapter")
    endif()
    nickel_domain_path(NICKEL_UI_SOURCE ui)
    set(NICKEL_UI_DOMAIN_INCLUDE_ROOT "${NICKEL_SHARED_NATIVE_ROOT}" CACHE PATH "Native domain include root for UI adapters")
    if(NOT NICKEL_UI_DOMAIN_INCLUDE_ROOT)
        set(NICKEL_UI_DOMAIN_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/native")
    endif()
    add_subdirectory("${NICKEL_UI_SOURCE}" "${CMAKE_BINARY_DIR}/ui")
    add_subdirectory(native/app)
    if(NICKEL_BUILD_TESTS)
        find_package(Qt6 6.8.3 EXACT CONFIG REQUIRED COMPONENTS Test)
        set(NICKEL_LAUNCH_REQUIRE_APP_PIPELINE ON CACHE BOOL "Require the production AppServices launch pipeline in root tests" FORCE)
        nickel_add_domain_tests(app app)
        nickel_domain_path(_nickel_app_test_source tests/app)
        if(NOT EXISTS "${_nickel_app_test_source}/production_tests.cmake")
            message(FATAL_ERROR "Missing canonical AppServices production test suite")
        endif()
        include("${_nickel_app_test_source}/production_tests.cmake")
        foreach(_action_domain app_instance_actions app_resource_actions app_launch_actions)
            nickel_add_domain_tests("${_action_domain}" "${_action_domain}")
        endforeach()
        target_compile_definitions(nickel_app_resource_actions_tests PRIVATE NICKEL_TRANSPORT_TEST_HOOKS=1)
        nickel_add_domain_tests(ui ui)
        nickel_add_domain_tests(brand brand)
        if(NOT EXISTS "${NICKEL_ENTRY_BUILD_INFO_DIRECTORY}/BuildInfo.hpp")
            message(FATAL_ERROR "Production entry tests require the actual generated application BuildInfo.hpp")
        endif()
        include("${CMAKE_CURRENT_SOURCE_DIR}/native/tests/production_entry/production_tests.cmake")
    endif()
endif()
include(cmake/NickelProductionSmoke.cmake)
